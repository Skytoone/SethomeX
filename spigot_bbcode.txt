[SIZE=5][B][COLOR=#FFD700]📄  CONFIGURATION OVERVIEW[/COLOR][/B][/SIZE]
[COLOR=#888888]────────────────────────────────────────────────────[/COLOR]
[COLOR=#AAAAAA]Generates a clean [B]config.yml[/B] with rich annotation support. Here is the logical structure:[/COLOR]

[SIZE=3] [/SIZE]

[CODE=yaml]
# -------------------------------------------------------------------------#
#                       SethomeX Configuration Engine                     #  
# -------------------------------------------------------------------------#

database:
  # Types supported: SQLITE (local) or MYSQL (multi-server sync)
  type: "SQLITE"
  mysql:
    host: "localhost"
    port: 3306
    database: "sethomex"
    pool:
      maximum-size: 10

teleport:
  # Warmup time in seconds before teleportation
  cooldown: 3
  cancel-on-move: true
  cancel-on-damage: true
  cancel-in-combat: true

# Home limits tied directly to group permissions: sethomex.limit.<group>
limits:
  default: 3
  vip: 5
  vip-plus: 10
  admin: 100

effects:
  countdown:
    # Supported: FLAME, ENCHANT, HAPPY_VILLAGER, WITCH, END_ROD, PORTAL, CHERRY_LEAVES
    particle: "PORTAL"
    # Visual Style: SPIRAL (ascending), RING (planar), SHIELD (spherical)
    style: "SPIRAL"
    sound: "BLOCK_NOTE_BLOCK_PLING"
  success:
    particle: "EXPLOSION"
    sound: "ENTITY_ENDERMAN_TELEPORT"

gui:
  title: "&e&lSethomeX &8» &fMes Habitations"
  default-home-item: "RED_BED"
  fill-empty-slots: true
  
  # Selection roster for the dynamic icon personalization system
  icon-selector:
    icons:
      - "RED_BED"
      - "COMPASS"
      - "GRASS_BLOCK"
      - "NETHERITE_SWORD"
      - "CAMPFIRE"
      - "DIAMOND"

safety:
  # Verifies landing zone safety (prevents lava, void, suffocation)
  check-safe-destination: true
  # Denies creating homes suspended in the air
  prevent-unsafe-sethome: true
[/CODE]
